COLS = 5760

PIXEL_SIZE_DEGREES = 0.0625
EQUATORIAL_PIXEL_SIZE_KM = 3.7044613374

print("\n" + "=" * 60)
print("TERRAFORM | MARS GLOBAL INTELLIGENCE")
print("=" * 60)

if not DATA_FILE.exists():
    raise FileNotFoundError(
        f"Mars dataset not found:\n{DATA_FILE}"
    )

actual_bytes = DATA_FILE.stat().st_size

if actual_bytes != EXPECTED_BYTES:
    raise ValueError(
        f"Unexpected file size. Expected {EXPECTED_BYTES:,} bytes, "
        f"received {actual_bytes:,} bytes."
    )

print("\n[1/5] Dataset file validated.")

# PDS label confirms signed 16-bit big-endian integers.

print("      Latitude range: -90° to +90°")

    "maximum_m": float(np.max(values)),
    "mean_m": float(np.mean(values)),
    "median_m": float(np.median(values)),
    "standard_deviation_m": float(np.std(values)),

        "p25": float(np.percentile(values, 25)),

"""
TERRAFORM — Global Planetary Terrain Comparison
Version: 1.0

Purpose:
Compare global elevation distributions for Earth, Mars and the Moon
using approximate common physical resolution and latitude-weighted
statistics.

Important:
- Earth uses an elevation-positive land proxy, not an authoritative land mask.
- Mars and Moon use their full gridded surfaces.
- Elevations retain their native planetary vertical references.
- This is a distribution comparison, NOT pixel-to-pixel matching.
"""

from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import rasterio
from rasterio.windows import Window
import matplotlib.pyplot as plt


# ============================================================
# 1. PROJECT PATHS AND CONFIGURATION
# ============================================================

ROOT = Path(__file__).resolve().parents[2]

EARTH_FILE = ROOT / "data" / "Global" / "Earth" / "earth_etopo_2022_60s.tif"
MARS_FILE = ROOT / "data" / "Global" / "Mars" / "megt90n000eb.img"
MOON_FILE = ROOT / "data" / "Global" / "Moon" / "ldem_16.img"

OUTPUT_DIR = ROOT / "outputs" / "Global" / "Comparisons"
VISUAL_DIR = ROOT / "outputs" / "Visualisations" / "Global"

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
VISUAL_DIR.mkdir(parents=True, exist_ok=True)

# Common elevation histogram:
# 10-metre bins, covering all observed planetary elevations.
ELEVATION_MIN = -11000
ELEVATION_MAX = 22000
ELEVATION_BIN_SIZE = 10

ELEVATION_EDGES = np.arange(
    ELEVATION_MIN,
    ELEVATION_MAX + ELEVATION_BIN_SIZE,
    ELEVATION_BIN_SIZE,
    dtype=np.float64,
)

ELEVATION_CENTRES = (
    ELEVATION_EDGES[:-1] + ELEVATION_EDGES[1:]
) / 2

# Process manageable blocks rather than loading Earth's full raster.
EARTH_OUTPUT_ROWS_PER_BLOCK = 128
PLANET_ROWS_PER_BLOCK = 128

# Approximate grid resolutions:
EARTH_SOURCE_ARC_MINUTES = 1.0
EARTH_AGGREGATION_FACTOR = 2

MARS_NATIVE_DEGREES = 0.0625
MOON_NATIVE_DEGREES = 0.0625
MOON_AGGREGATION_FACTOR = 2


# ============================================================
# 2. SHARED STATISTICAL FUNCTIONS
# ============================================================

def latitude_weights(
    first_row: int,
    number_of_rows: int,
    resolution_degrees: float,
) -> np.ndarray:
    """
    Approximate spherical cell-area weights.

    For a regular latitude-longitude grid, cell area is proportional
    to cos(latitude). The planet's radius is constant within each
    planetary dataset and therefore cancels in normalized statistics.

    This is an approximation, not a full geodesic area calculation.
    """

    row_indices = np.arange(
        first_row,
        first_row + number_of_rows,
        dtype=np.float64,
    )

    latitude_centres = (
        90.0 - (row_indices + 0.5) * resolution_degrees
    )

    weights = np.cos(np.deg2rad(latitude_centres))

    return np.clip(weights, 0.0, None)


def add_to_histogram(
    histogram: np.ndarray,
    values: np.ndarray,
    weights: np.ndarray,
) -> None:
    """Add a weighted block to an existing histogram."""

    values = np.asarray(values).ravel()
    weights = np.asarray(weights).ravel()

    valid = (
        np.isfinite(values)
        & np.isfinite(weights)
        & (weights > 0)
        & (values >= ELEVATION_MIN)
        & (values < ELEVATION_MAX)
    )

    if not np.any(valid):
        return

    block_histogram, _ = np.histogram(
        values[valid],
        bins=ELEVATION_EDGES,
        weights=weights[valid],
    )

    histogram += block_histogram


def calculate_weighted_statistics(
    histogram: np.ndarray,
) -> dict:
    """
    Calculate approximate weighted statistics from histogram bins.

    Percentiles and moments are approximated using 10-metre bins.
    """

    total_weight = float(histogram.sum())

    if total_weight <= 0:
        raise ValueError("No valid weighted observations were found.")

    probabilities = histogram / total_weight

    mean = float(np.sum(probabilities * ELEVATION_CENTRES))

    variance = float(
        np.sum(
            probabilities
            * (ELEVATION_CENTRES - mean) ** 2
        )
    )

    standard_deviation = float(np.sqrt(max(variance, 0.0)))

    cumulative = np.cumsum(histogram)

    def weighted_percentile(percentile: float) -> float:
        target = total_weight * percentile / 100.0

        index = int(np.searchsorted(cumulative, target))
        index = min(index, len(histogram) - 1)

        previous = (
            float(cumulative[index - 1])
            if index > 0
            else 0.0
        )

        bin_weight = float(histogram[index])

        if bin_weight <= 0:
            fraction = 0.0
        else:
            fraction = (target - previous) / bin_weight

        fraction = float(np.clip(fraction, 0.0, 1.0))

        return float(
            ELEVATION_EDGES[index]
            + fraction * ELEVATION_BIN_SIZE
        )

    percentiles = {
        f"P{p:02d}": weighted_percentile(p)
        for p in (5, 10, 25, 50, 75, 90, 95)
    }

    return {
        "weighted_mean_m": mean,
        "weighted_standard_deviation_m": standard_deviation,
        "weighted_percentiles_m": percentiles,
        "weighted_histogram_total": total_weight,
        "histogram_bin_width_m": ELEVATION_BIN_SIZE,
    }


def make_normalized_histogram(
    histogram: np.ndarray,
    mean: float,
    standard_deviation: float,
) -> tuple[np.ndarray, np.ndarray]:
    """Convert elevation histogram into a standardized z-score histogram."""

    z_edges = np.linspace(-6, 6, 241)
    z_centres = (z_edges[:-1] + z_edges[1:]) / 2

    if standard_deviation == 0:
        raise ValueError("Standard deviation is zero; cannot normalize.")

    z_values = (
        ELEVATION_CENTRES - mean
    ) / standard_deviation

    normalized_histogram, _ = np.histogram(
        z_values,
        bins=z_edges,
        weights=histogram,
    )

    total = float(histogram.sum())

    if total > 0:
        normalized_histogram = normalized_histogram / total

    return z_centres, normalized_histogram

def process_earth() -> dict:
    """
    Process ETOPO Earth data.

    Native resolution: 1 arc-minute.
    Aggregation: 2 x 2 cells -> 2 arc-minutes.

    Land proxy:
    A coarse cell is retained when at least 2 of its 4 original
    cells have valid elevation > 0 metres.

    The mean is calculated using only those positive cells.
    This is a transparent elevation-positive land proxy, not
    a definitive coastline or land-cover mask.
    """

    print("\n[1/3] Processing Earth...")

    if not EARTH_FILE.exists():
        raise FileNotFoundError(
            f"Earth dataset not found: {EARTH_FILE}"
        )

    histogram = np.zeros(
        len(ELEVATION_EDGES) - 1,
        dtype=np.float64,
    )

    accepted_cells = 0
    total_coarse_cells = 0

    with rasterio.open(EARTH_FILE) as src:

        expected_width = 21600
        expected_height = 10800

        if (
            src.width != expected_width
            or src.height != expected_height
        ):
            raise ValueError(
                "Unexpected Earth raster dimensions. "
                f"Found {src.width} x {src.height}; "
                f"expected {expected_width} x {expected_height}."
            )

        output_width = src.width // EARTH_AGGREGATION_FACTOR
        output_height = src.height // EARTH_AGGREGATION_FACTOR

        for output_start in range(
            0,
            output_height,
            EARTH_OUTPUT_ROWS_PER_BLOCK,
        ):

            output_count = min(
                EARTH_OUTPUT_ROWS_PER_BLOCK,
                output_height - output_start,
            )

            source_start = output_start * 2
            source_count = output_count * 2

            masked_block = src.read(
                1,
                window=Window(
                    0,
                    source_start,
                    src.width,
                    source_count,
                ),
                masked=True,
            )

            raw = np.asarray(
                masked_block.data,
                dtype=np.float32,
            )

            mask = np.ma.getmaskarray(masked_block)

            valid = np.isfinite(raw) & (~mask)
            positive = valid & (raw > 0)

            # Group each 2 x 2 source-cell block.
            grouped_values = raw.reshape(
                output_count,
                2,
                output_width,
                2,
            )

            grouped_positive = positive.reshape(
                output_count,
                2,
                output_width,
                2,
            )

            positive_count = grouped_positive.sum(
                axis=(1, 3)
            )

            positive_sum = np.where(
                grouped_positive,
                grouped_values,
                0.0,
            ).sum(axis=(1, 3), dtype=np.float64)

            # Retain cells with at least 2 positive source cells.
            keep = positive_count >= 2

            coarse_values = np.zeros(
                positive_count.shape,
                dtype=np.float32,
            )

            np.divide(
                positive_sum,
                positive_count,
                out=coarse_values,
                where=keep,
            )

            row_weights = latitude_weights(
                output_start,
                output_count,
                resolution_degrees=2.0 / 60.0,
            )

            cell_weights = np.broadcast_to(
                row_weights[:, None],
                coarse_values.shape,
            )

            add_to_histogram(
                histogram,
                coarse_values[keep],
                cell_weights[keep],
            )

            accepted_cells += int(keep.sum())
            total_coarse_cells += int(keep.size)

            if output_start % 512 == 0:
                print(
                    f"  Earth progress: "
                    f"{output_start:,}/{output_height:,} rows"
                )

    statistics = calculate_weighted_statistics(histogram)

    print("  Earth processing complete.")

    return {
        "planet": "Earth",
        "dataset": "NOAA ETOPO 2022, 60 arc-second",
        "input_file": str(EARTH_FILE.relative_to(ROOT)),
        "native_resolution_degrees": 1.0 / 60.0,
        "comparison_resolution_degrees": 2.0 / 60.0,
        "approximate_equatorial_north_south_spacing_km": 3.7,
        "surface_scope": "Elevation-positive land proxy",
        "land_proxy_rule": (
            "At least 2 of 4 original 1-arc-minute cells "
            "must have valid elevation > 0 m. "
            "Mean uses positive cells only."
        ),
        "accepted_coarse_cells": accepted_cells,
        "total_coarse_grid_cells": total_coarse_cells,
        "accepted_grid_cell_fraction": (
            accepted_cells / total_coarse_cells
            if total_coarse_cells
            else None
        ),
        "vertical_reference": "WGS84 + EGM2008 geoid",
        "statistics": statistics,
        "histogram": histogram,
        "limitations": [
            "Elevation > 0 m is a preliminary land proxy, not an authoritative land mask.",
            "Below-sea-level terrestrial regions may be excluded.",
            "Coastal blocks are handled using a 2-of-4 positive-cell threshold.",
            "Earth is represented by a land proxy, not its full land-and-ocean surface.",
            "Latitude weighting uses a spherical cos(latitude) approximation.",
        ],
    }

def process_mars() -> dict:
    """
    Process MOLA MEGDR global elevation.

    Native resolution: 0.0625 degrees.
    Data type: big-endian signed 16-bit integer.
    Units: metres relative to MOLA areoid.
    """

    print("\n[2/3] Processing Mars...")

    if not MARS_FILE.exists():
        raise FileNotFoundError(
            f"Mars dataset not found: {MARS_FILE}"
        )

    rows = 2880
    columns = 5760
    expected_cells = rows * columns

    raw = np.fromfile(
        MARS_FILE,
        dtype=">i2",
    )

    if raw.size != expected_cells:
        raise ValueError(
            "Mars raster size mismatch. "
            f"Found {raw.size:,} cells; "
            f"expected {expected_cells:,}."
        )

    elevation = raw.reshape(rows, columns).astype(
        np.float32
    )

    histogram = np.zeros(
        len(ELEVATION_EDGES) - 1,
        dtype=np.float64,
    )

    for start in range(0, rows, PLANET_ROWS_PER_BLOCK):

        end = min(
            start + PLANET_ROWS_PER_BLOCK,
            rows,
        )

        block = elevation[start:end]

        row_weights = latitude_weights(
            start,
            end - start,
            resolution_degrees=MARS_NATIVE_DEGREES,
        )

        weights = np.broadcast_to(
            row_weights[:, None],
            block.shape,
        )

        add_to_histogram(
            histogram,
            block,
            weights,
        )

        if start % 512 == 0:
            print(
                f"  Mars progress: {start:,}/{rows:,} rows"
            )

    statistics = calculate_weighted_statistics(histogram)

    print("  Mars processing complete.")

    return {
        "planet": "Mars",
        "dataset": "MOLA MEGDR MEGT90N000EB",
        "input_file": str(MARS_FILE.relative_to(ROOT)),
        "native_resolution_degrees": MARS_NATIVE_DEGREES,
        "comparison_resolution_degrees": MARS_NATIVE_DEGREES,
        "approximate_equatorial_north_south_spacing_km": 3.704,
        "surface_scope": "Full gridded planetary surface",
        "aggregation": "Native grid retained",
        "grid_dimensions": {
            "rows": rows,
            "columns": columns,
        },
        "vertical_reference": "MOLA areoid",
        "data_interpretation": (
            "Median observed topography where measurements exist; "
            "interpolated values are also present."
        ),
        "zero_values": "Retained as valid elevation values",
        "statistics": statistics,
        "histogram": histogram,
        "limitations": [
            "MOLA measurement coverage is spatially uneven.",
            "Interpolated cells are included in the global distribution.",
            "Grid-cell statistics are weighted using cos(latitude).",
            "The spherical area-weighting approximation is not a full geodesic calculation.",
        ],
    }

def process_moon() -> dict:
    """
    Process LOLA LDEM_16 global elevation.

    Native resolution: 0.0625 degrees.
    Aggregation: 2 x 2 cells -> 0.125 degrees.
    Data type: little-endian signed 16-bit integer.
    Elevation: DN * 0.5 metres.
    """

    print("\n[3/3] Processing Moon...")

    if not MOON_FILE.exists():
        raise FileNotFoundError(
            f"Moon dataset not found: {MOON_FILE}"
        )

    rows = 2880
    columns = 5760
    expected_cells = rows * columns

    raw = np.fromfile(
        MOON_FILE,
        dtype="<i2",
    )

    if raw.size != expected_cells:
        raise ValueError(
            "Moon raster size mismatch. "
            f"Found {raw.size:,} cells; "
            f"expected {expected_cells:,}."
        )

    # Convert digital numbers to elevation in metres.
    elevation = raw.reshape(
        rows,
        columns,
    ).astype(np.float32) * 0.5

    # Aggregate 2 x 2 cells.
    coarse_elevation = elevation.reshape(
        rows // 2,
        2,
        columns // 2,
        2,
    ).mean(axis=(1, 3))

    coarse_rows, coarse_columns = coarse_elevation.shape

    histogram = np.zeros(
        len(ELEVATION_EDGES) - 1,
        dtype=np.float64,
    )

    moon_resolution = MOON_NATIVE_DEGREES * 2

    for start in range(
        0,
        coarse_rows,
        PLANET_ROWS_PER_BLOCK,
    ):

        end = min(
            start + PLANET_ROWS_PER_BLOCK,
            coarse_rows,
        )

        block = coarse_elevation[start:end]

        row_weights = latitude_weights(
            start,
            end - start,
            resolution_degrees=moon_resolution,
        )

        weights = np.broadcast_to(
            row_weights[:, None],
            block.shape,
        )

        add_to_histogram(
            histogram,
            block,
            weights,
        )

        if start % 256 == 0:
            print(
                f"  Moon progress: "
                f"{start:,}/{coarse_rows:,} rows"
            )

    statistics = calculate_weighted_statistics(histogram)

    print("  Moon processing complete.")

    return {
        "planet": "Moon",
        "dataset": "NASA LOLA LDEM_16",
        "input_file": str(MOON_FILE.relative_to(ROOT)),
        "native_resolution_degrees": MOON_NATIVE_DEGREES,
        "comparison_resolution_degrees": moon_resolution,
        "approximate_equatorial_north_south_spacing_km": 3.79,
        "surface_scope": "Full gridded lunar surface",
        "aggregation": "2 x 2 native cells averaged",
        "grid_dimensions": {
            "rows": coarse_rows,
            "columns": coarse_columns,
        },
        "vertical_reference": (
            "Reference sphere with radius 1,737.4 km"
        ),
        "height_conversion": "Elevation metres = DN * 0.5",
        "zero_values": "Retained as valid elevation values",
        "statistics": statistics,
        "histogram": histogram,
        "limitations": [
            "LOLA grid interpolation is used to compensate for incomplete altimetry coverage.",
            "The product has documented edge artefacts near some 45-degree latitude bands.",
            "The vertical reference is a reference sphere, not a geoid or equipotential surface.",
            "The known raw-extrema versus label-extrema discrepancy remains unresolved.",
            "Latitude weighting uses a spherical cos(latitude) approximation.",
        ],
    }

def create_elevation_distribution_chart(
    results: dict,
) -> Path:

    fig, ax = plt.subplots(figsize=(13, 7))

    display_order = ["Earth", "Mars", "Moon"]

    for name in display_order:

        item = results[name]
        histogram = item["histogram"]

        total = histogram.sum()

        if total <= 0:
            continue

        density = histogram / total / ELEVATION_BIN_SIZE

        ax.plot(
            ELEVATION_CENTRES,
            density,
            linewidth=2,
            label=name,
        )

    ax.set_title(
        "Global Planetary Elevation Distributions",
        fontsize=17,
        pad=18,
    )

    ax.set_xlabel(
        "Elevation relative to each dataset's native vertical reference (m)",
        fontsize=11,
    )

    ax.set_ylabel(
        "Latitude-weighted relative frequency per metre",
        fontsize=11,
    )

    ax.grid(
        True,
        alpha=0.25,
    )

    ax.legend(
        frameon=False,
    )

    fig.text(
        0.5,
        0.015,
        "Earth: elevation-positive land proxy | "
        "Mars and Moon: full gridded surfaces | "
        "Native vertical references differ",
        ha="center",
        fontsize=9,
    )

    fig.tight_layout(rect=[0, 0.05, 1, 1])

    output = VISUAL_DIR / "global_elevation_distribution_comparison.png"

    fig.savefig(
        output,
        dpi=220,
        bbox_inches="tight",
    )

    plt.close(fig)

    return output

def clean_for_json(value):
    """Convert NumPy values and arrays into JSON-safe Python values."""

    if isinstance(value, dict):
        return {
            key: clean_for_json(item)
            for key, item in value.items()
            if key != "histogram"
        }

    if isinstance(value, (list, tuple)):
        return [
            clean_for_json(item)
            for item in value
        ]

    if isinstance(value, np.ndarray):
        return value.tolist()

    if isinstance(value, (np.integer,)):
        return int(value)

    if isinstance(value, (np.floating,)):
        return float(value)

    if isinstance(value, (np.bool_,)):
        return bool(value)

    return value
