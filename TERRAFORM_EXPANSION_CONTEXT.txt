def validate_catalog(datasets):
    required_fields = [
        "dataset_id",
        "planet",
        "dataset_name",
        "provider",
        "spatial_coverage",
        "measurement",
        "measurement_unit",
        "source_url",
        "processing_status"
    ]

    errors = []

    for dataset in datasets:
        for field in required_fields:
            if not dataset.get(field):
                errors.append(
                    f"{dataset.get('dataset_id', 'UNKNOWN')}: "
                    f"missing {field}"
                )

    identifiers = [item["dataset_id"] for item in datasets]

    if len(identifiers) != len(set(identifiers)):
        errors.append("Duplicate dataset identifiers detected.")

    return errors


def build_catalog():
    errors = validate_catalog(DATASETS)

    if errors:
        raise ValueError("\n".join(errors))

    return {
        "project": "TERRAFORM",
        "project_title": "Planetary Analog Intelligence",
        "catalog_version": "1.0",
        "generated_utc": datetime.now(timezone.utc).isoformat(),
        "dataset_count": len(DATASETS),
        "planets": ["Earth", "Mars", "Moon"],
        "datasets": DATASETS,
        "methodological_note": (
            "Dataset resolutions, coordinate systems, vertical references "
            "and measurement conventions must be verified before "
            "cross-planetary comparison. Global screening and regional "
            "analysis are separate processing levels."
        )
    }

"""
TERRAFORM - Dashboard Data Integration
Connects uploaded CSV files to the validation and processing backend.

from pathlib import Path
import pandas as pd

from src.data.processing import process_dataframe


PROJECT_ROOT = Path(__file__).resolve().parents[2]

EARTH_RAW = PROJECT_ROOT / "data" / "raw" / "earth"
MARS_RAW = PROJECT_ROOT / "data" / "raw" / "mars"

EARTH_RAW.mkdir(parents=True, exist_ok=True)
MARS_RAW.mkdir(parents=True, exist_ok=True)


def load_and_process(uploaded_file, dataset_type):
    """
    Load a CSV from the dashboard or raw data directory,
    validate it, and return cleaned/rejected records and report.
    """

    dataset_type = dataset_type.lower()

    if dataset_type not in ("earth", "mars"):
        raise ValueError("Dataset type must be earth or mars.")

    raw_directory = EARTH_RAW if dataset_type == "earth" else MARS_RAW

    if uploaded_file is not None:

        dataframe = pd.read_csv(uploaded_file)
        source = "Dashboard upload"

    else:

        raw_files = sorted(raw_directory.glob("*.csv"))

        if not raw_files:

            return {
                "cleaned": pd.DataFrame(),
                "rejected": pd.DataFrame(),
                "report": None,
                "source": "No dataset available",
                "saved_paths": {},
            }

        selected_file = raw_files[0]

        dataframe = pd.read_csv(selected_file)
        source = f"Local file: {selected_file.name}"

    result = process_dataframe(
        dataframe=dataframe,
        dataset_type=dataset_type,
        save=True
    )

    return {
        "cleaned": result["cleaned"],
        "rejected": result["rejected"],
        "report": result["report"],
        "source": source,
        "saved_paths": result["saved_paths"],
    }

"""
TERRAFORM - Data Processing Pipeline
Version: 1.0
"""

from pathlib import Path
import json
import pandas as pd

from src.data.validation import validate_dataset


PROJECT_ROOT = Path(__file__).resolve().parents[2]

PROCESSED_DIR = PROJECT_ROOT / "data" / "processed"
REPORT_DIR = PROJECT_ROOT / "outputs" / "reports"

PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
REPORT_DIR.mkdir(parents=True, exist_ok=True)


def process_dataframe(dataframe, dataset_type, save=True):
    """
    Validate and process a pandas DataFrame.

    Returns a dictionary containing:
    - cleaned dataset
    - rejected records
    - validation report
    - saved file paths
    """

    result = validate_dataset(
        dataframe,
        dataset_type
    )

    cleaned = result["cleaned"]
    rejected = result["rejected"]
    report = result["report"]

    saved_paths = {}

    if save and report["status"] == "PASSED":

        processed_path = (
            PROCESSED_DIR
            / f"{dataset_type}_processed.csv"
        )

        rejected_path = (
            PROCESSED_DIR
            / f"{dataset_type}_rejected.csv"
        )

        report_path = (
            REPORT_DIR
            / f"{dataset_type}_validation_report.json"
        )

        cleaned.to_csv(
            processed_path,
            index=False
        )

        rejected.to_csv(
            rejected_path,
            index=False
        )

        with open(
            report_path,
            "w",
            encoding="utf-8"
        ) as report_file:

            json.dump(
                report,
                report_file,
                indent=4,
                ensure_ascii=False
            )

        saved_paths = {
            "processed_data": str(processed_path),
            "rejected_data": str(rejected_path),
            "validation_report": str(report_path),
        }

    return {
        "cleaned": cleaned,
        "rejected": rejected,
        "report": report,
        "saved_paths": saved_paths,
    }
