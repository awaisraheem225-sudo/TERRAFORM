def visualisations_page():
    section_header(
        "MODULE 05 / SCIENTIFIC VISUALISATION",
        "Research Visualisations",
        "A visual archive of the global planetary analysis and the Atacama–Jezero regional investigation."
    )

    total_images = sum(len(items) for items in VISUALISATION_GROUPS.values())
    available_images = sum(
        (OUTPUTS / path).exists()
        for items in VISUALISATION_GROUPS.values()
        for _, path in items
    )

    a, b, c = st.columns(3)
    a.metric("Visual assets catalogued", total_images)
    b.metric("Images available", available_images)
    c.metric("Missing images", total_images - available_images)

    st.caption(
        "The gallery uses the existing image files in the outputs directory. "
        "No image is regenerated or altered by this page."
    )
    st.divider()

    for group_name, images in VISUALISATION_GROUPS.items():
        st.markdown(f"### {group_name}")
        cols = st.columns(2)
        for index, (title, relative_path) in enumerate(images):
            with cols[index % 2]:
                st.markdown(f"**{title}**")
                show_image(relative_path, title)
        st.divider()

    st.markdown("### Visualisation inventory")
    inventory_rows = []
    for group_name, images in VISUALISATION_GROUPS.items():
        for title, relative_path in images:
            path = OUTPUTS / relative_path
            inventory_rows.append({
                "Research stream": group_name,
                "Visualisation": title,
                "Status": "Available" if path.exists() else "Missing",
                "Path": f"outputs/{relative_path}",
            })
    st.dataframe(
        pd.DataFrame(inventory_rows),
        use_container_width=True,
        hide_index=True
    )
    st.download_button(
        "Download visualisation inventory",
        data=pd.DataFrame(inventory_rows).to_csv(index=False).encode("utf-8"),
        file_name="terraform_visualisation_inventory.csv",
        mime="text/csv"
    )
